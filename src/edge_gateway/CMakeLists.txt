cmake_minimum_required(VERSION 3.16)
project(edge_gateway)

find_package(ament_cmake REQUIRED)
find_package(edge_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_library(JSON_C_LIBRARY json-c REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(std_msgs REQUIRED)

add_executable(edge_gateway_node src/edge_gateway_node.cpp)
target_compile_features(edge_gateway_node PRIVATE cxx_std_20)
target_compile_options(edge_gateway_node PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(edge_gateway_node edge_interfaces geometry_msgs nav_msgs sensor_msgs rclcpp rclcpp_action std_msgs)
target_link_libraries(edge_gateway_node ${JSON_C_LIBRARY})
install(TARGETS edge_gateway_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(body_authority_test test/body_authority_test.cpp)
  target_compile_features(body_authority_test PRIVATE cxx_std_20)
  target_include_directories(body_authority_test PRIVATE src)
  add_test(NAME body_authority_contract COMMAND body_authority_test)
endif()
ament_package()
