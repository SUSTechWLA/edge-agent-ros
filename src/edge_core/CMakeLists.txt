cmake_minimum_required(VERSION 3.16)
project(edge_core)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(edge_interfaces REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(SQLite3 REQUIRED)
add_executable(edge_core_node src/core_node.cpp src/ledger.cpp)
target_compile_features(edge_core_node PRIVATE cxx_std_20)
target_include_directories(edge_core_node PRIVATE include)
target_compile_options(edge_core_node PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(edge_core_node rclcpp rclcpp_action edge_interfaces nav_msgs std_msgs)
target_link_libraries(edge_core_node SQLite::SQLite3)
install(TARGETS edge_core_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(ledger_test test/ledger_test.cpp src/ledger.cpp)
  target_compile_features(ledger_test PRIVATE cxx_std_20)
  target_include_directories(ledger_test PRIVATE src include)
  target_link_libraries(ledger_test PRIVATE SQLite::SQLite3)
  add_test(NAME ledger_persistence COMMAND ledger_test)
  add_executable(policy_test test/policy_test.cpp)
  target_compile_features(policy_test PRIVATE cxx_std_20)
  target_include_directories(policy_test PRIVATE include)
  add_test(NAME policy_contract COMMAND policy_test)
endif()
ament_package()
